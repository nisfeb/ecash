const API = "/apps/ecash-services/admin/api";

// ── Fetch helper: same-origin cookie, JSON parse, surface non-200 detail ──
async function req(path, opts) {
  opts = opts || {};
  const headers = opts.body ? {'Content-Type': 'application/json', ...(opts.headers || {})} : (opts.headers || {});
  const r = await fetch(API + path, {credentials: 'same-origin', ...opts, headers});
  let data;
  try { data = await r.json(); } catch { data = null; }
  if (!r.ok) {
    const detail = data && (data.detail || data.error || data.message);
    throw new Error(detail || ('HTTP ' + r.status));
  }
  return data;
}
const post = (path, body) => req(path, {method: 'POST', body: JSON.stringify(body)});
const run = (p, then) => p.then(then).catch(e => toast(e.message));

// ── Misc helpers ─────────────────────────────────────────────
const el = id => document.getElementById(id);
const val = id => el(id).value.trim();
const guard = (msg, fn) => { if (confirm(msg)) fn(); };

function toast(msg) {
  const t = el("toast");
  t.textContent = msg || 'Copied!';
  t.style.opacity = "1";
  setTimeout(() => { t.style.opacity = "0"; }, 2200);
}

// navigator.clipboard exists only in a secure context, so not on a
// plain-HTTP LAN address: fall back to a prompt to copy from.
function copyText(t) {
  const manual = () => prompt('Copy:', t);
  if (navigator.clipboard) navigator.clipboard.writeText(t).then(() => toast('Copied!'), manual);
  else manual();
}

const esc = (s) => String(s == null ? '' : s)
  .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
  .replace(/'/g, '&#39;').replace(/"/g, '&quot;');

function fmtDate(unix) {
  if (!unix) return '?';
  return new Date(unix * 1000).toLocaleString();
}

function box(val, label) {
  const v = esc(val);
  return '<div class="box"><div class="num ' + (v.length > 16 ? 'sm' : '') + '">' + v + '</div><div class="lbl">' + esc(label) + '</div></div>';
}

function setStatus(ok) {
  const s = el("sta");
  s.textContent = ok ? "Online" : "Error";
  s.style.color = ok ? "#5eead4" : "#fca5a5";
}

// datetime-local value (local time) <-> unix seconds
function toLocalInput(unix) {
  try {
    const d = new Date(unix * 1000);
    d.setMinutes(d.getMinutes() - d.getTimezoneOffset());
    return d.toISOString().slice(0, 16);
  } catch { return ''; }
}

// Optional whole-number fields: blank is null (none), anything that is not
// a whole number >= 0 throws, so bad input never reaches the server.
function optInt(input, what) {
  const v = input.value.trim();
  if (input.validity.badInput) throw new Error(what + ': enter a whole number');
  if (v === '') return null;
  const n = Number(v);
  if (!Number.isInteger(n) || n < 0) throw new Error(what + ': enter a whole number >= 0');
  return n;
}
function optTime(input) {
  if (input.validity.badInput) throw new Error('Expires: incomplete date');
  if (input.value === '') return null;
  const n = Math.floor(new Date(input.value).getTime() / 1000);
  if (!Number.isInteger(n) || n < 0) throw new Error('Expires: invalid date');
  return n;
}

let _ar = null, _arAt = 0;
function startAuto(loader, sec) {
  stopAuto();
  if (!el('autorefresh').checked) return;
  _ar = setInterval(() => loader(true), (sec || 10) * 1000);
}
function stopAuto() { if (_ar) { clearInterval(_ar); _ar = null; } }
function stampUpdated() {
  _arAt = Date.now();
  el('updated').textContent = 'updated just now';
}
setInterval(() => {
  if (_arAt) el('updated').textContent = `updated ${Math.round((Date.now() - _arAt) / 1000)}s ago`;
}, 1000);

// ── Tab switching ────────────────────────────────────────────
document.querySelectorAll("#main-tabs .tab").forEach(btn => {
  btn.addEventListener("click", () => {
    const t = btn.dataset.tab;
    document.querySelectorAll("#main-tabs .tab").forEach(x => x.classList.remove("on"));
    document.querySelectorAll(".sec").forEach(s => s.classList.remove("on"));
    btn.classList.add("on");

    else { stopAuto(); }
    if (t === 'sv') loadServices();

  });

function capLine(s) {
  if (s.max_issuance == null) return {line: 'issued <b>' + esc(s.issued) + '</b>', gauge: ''};
  const pct = s.max_issuance > 0 ? Math.min(100, Math.round((s.issued / s.max_issuance) * 100)) : 0;
  return {
    line: '<b>' + esc(s.issued) + ' / ' + esc(s.max_issuance) + '</b> (' + pct + '%)',
    gauge: '<div class="gauge"><div class="gauge-fill" style="width:' + pct + '%"></div></div>',
  };
}

// ── Overview tab (read-only, auto-refresh) ───────────────────
async function loadOverview(silent) {
  try {
    const [cred, svc] = await Promise.all([req('/cred/overview'), req('/services')]);
    const services = svc.services || [];
    const activeCount = services.filter(s => s.active).length;
    const totalIssued = services.reduce((a, s) => a + (s.issued || 0), 0);
    const totalRedeemed = services.reduce((a, s) => a + (s.redeemed || 0), 0);

    el("ov-stats").innerHTML =
      box(activeCount + ' / ' + services.length, "Services active") +
      box(cred.cred_keysets || 0, "Cred keysets") +
      box(cred.cred_issued || 0, "Credentials issued") +
      box(cred.cred_spent || 0, "Credentials spent") +
      box(totalIssued, "Total issued") +
      box(totalRedeemed, "Total redeemed");

    el("ov-gauges").innerHTML = !services.length
      ? '<div class="empty">No services yet.</div>'
      : services.map(s => {
          const c = capLine(s);
          return '<div style="margin-bottom:12px">' +
            '<div style="display:flex;justify-content:space-between;font-size:.82rem">' +
              '<span class="accent">' + esc(s.name) + '</span>' +
              '<span class="muted">' + c.line + '</span>' +
            '</div>' + c.gauge +
          '</div>';
        }).join('');
    stampUpdated();
    setStatus(true);
  } catch(e) {
    setStatus(false);
    if (!silent) toast(e.message);
  }
}

el("autorefresh").addEventListener("change", e => {
  if (e.target.checked) startAuto(loadOverview); else stopAuto();
});

      : '<span class="badge b-public">public</span>',

        '<div class="fld"><label for="' + id + '-desc">Description</label><input id="' + id + '-desc" class="ed-desc" value="' + esc(s.description) + '"></div>' +
      '</div>' +
      '<div class="frow">' +
        '<div class="fld"><label for="' + id + '-max">Max issuance (blank = unlimited)</label>' +
          '<input id="' + id + '-max" class="ed-max" type="number" min="0" step="1" placeholder="unlimited" value="' + esc(s.max_issuance) + '">' +

          '<input id="' + id + '-exp" class="ed-expires" type="datetime-local" value="' + (s.expires != null ? esc(toLocalInput(s.expires)) : '') + '">' +

    listEl.innerHTML = '<div class="empty">Error: ' + esc(e.message) + '</div>';
  }
}

async function createService() {
  try {
    const name = val('svc-name');
    if (!/^[a-z0-9_-]{1,64}$/.test(name) || name === 'list')
      throw new Error('Name: 1-64 of a-z, 0-9, _ or - (and not "list")');
    await post('/services/create', {
      name, title: val('svc-title'), description: val('svc-desc'),
      max_issuance: optInt(el('svc-max'), 'Max issuance'),
      expires: optTime(el('svc-expires')),
    });
    ['svc-name', 'svc-title', 'svc-desc', 'svc-expires', 'svc-max'].forEach(id => { el(id).value = ''; });
    loadServices();
  } catch(e) { toast(e.message); }
}

// Send only the fields that differ from what the card was rendered with.
function saveEdit(btn) {
  const card = btn.closest('.svc-card');
  const q = sel => card.querySelector(sel);
  const changed = sel => q(sel).value !== q(sel).defaultValue || q(sel).validity.badInput;
  try {
    const patch = {name: btn.dataset.name};
    if (changed('.ed-title')) {
      patch.title = q('.ed-title').value.trim();
      if (!patch.title) throw new Error('Title is required');
    }
    if (changed('.ed-desc')) patch.description = q('.ed-desc').value;
    if (changed('.ed-max')) patch.max_issuance = optInt(q('.ed-max'), 'Max issuance');
    if (changed('.ed-expires')) patch.expires = optTime(q('.ed-expires'));
    run(post('/services/update', patch), loadServices);
  } catch(e) { toast(e.message); }
}

function addKeyForCard(btn) {
  const key = btn.closest('.svc-card').querySelector('.al-input').value.trim();
  if (!key) { toast('Enter a key first.'); return; }
  run(post('/services/allowlist/add', {name: btn.dataset.name, key}), loadServices);
}

// ── Credentials tab (raw cred keysets) ───────────────────────
async function loadCreds() {
  const listEl = el("cr-list");
  try {
    const d = await req('/cred/overview');
    el("cr-stats").innerHTML =
      box(d.cred_keysets || 0, "Cred keysets") +
      box(d.cred_issued || 0, "Credentials issued") +
      box(d.cred_spent || 0, "Credentials spent");
    setStatus(true);

    // plain keysets first; a service's keyset is managed by its service
    const keysets = (d.keysets || []).sort((a, b) => !!a.service_scoped - !!b.service_scoped);
    if (!keysets.length) {
      listEl.innerHTML = '<div class="empty">No credential keysets. Use Generate to create one.</div>';
      return;
    }
    listEl.innerHTML =
      '<div class="tbl-wrap"><table class="tbl"><thead><tr><th>Keyset ID</th><th>Status</th><th>Actions</th></tr></thead><tbody>' +
      keysets.map(k => {
        const id = esc(k.id);
        const owner = k.service_scoped
          ? ' <span class="badge b-svc">' + (k.service ? 'service: ' + esc(k.service) : 'deleted service') + '</span>'
          : '';
        const action = k.service_scoped
          ? '<span class="muted">managed by its service</span>'
          : k.active
            ? '<button class="rbtn" data-action="cred-deact" data-id="' + id + '">Deactivate</button>'
            : '<button class="rbtn" data-action="cred-act" data-id="' + id + '">Activate</button>';
        return '<tr>' +
          '<td class="mono accent"><button class="linkbtn" data-action="copy" data-copy="' + id + '" title="Copy keyset id">' + id + '</button></td>' +
          '<td>' + (k.active ? '<span class="badge b-active">active</span>' : '<span class="badge b-inactive">inactive</span>') + owner + '</td>' +
          '<td style="white-space:nowrap">' + action + '</td>' +
        '</tr>';
      }).join('') +
      '</tbody></table></div>';
  } catch(e) {
    setStatus(false);
    listEl.innerHTML = '<div class="empty">Error: ' + esc(e.message) + '</div>';
  }
}

// ── Every button acts through data-action (the CSP forbids inline handlers) ──
const actions = {
  'refresh-ov': () => loadOverview(),
  'refresh-sv': () => loadServices(),
  'refresh-cr': () => loadCreds(),
  'create-svc': () => createService(),
  'gen-cred':   () => run(req('/cred/keysets/generate', {method: 'POST'}), loadCreds),
  'cred-act':   b => run(post('/cred/keysets/activate', {id: b.dataset.id}), loadCreds),
  'cred-deact': b => guard(`Deactivate keyset ${b.dataset.id}? No credentials can be issued from it until it is activated again.`,
                  () => run(post('/cred/keysets/deactivate', {id: b.dataset.id}), loadCreds)),
  'svc-act':    b => run(post('/services/activate', {name: b.dataset.name}), loadServices),
  'svc-deact':  b => guard(`Deactivate "${b.dataset.name}"?`,
                  () => run(post('/services/deactivate', {name: b.dataset.name}), loadServices)),
  'svc-del':    b => guard(`Delete "${b.dataset.name}"? Only allowed if inactive and never issued.`,
                  () => run(post('/services/delete', {name: b.dataset.name}), loadServices)),
  'svc-edit':   b => { const ed = b.closest('.svc-card').querySelector('.svc-edit'); ed.hidden = !ed.hidden; },
  'svc-save':   b => saveEdit(b),
  'key-add':    b => addKeyForCard(b),
  'key-remove': b => guard('Revoke this access key?',
                  () => run(post('/services/allowlist/remove', {name: b.dataset.name, key: b.dataset.key}), loadServices)),
  'copy':       b => copyText(b.dataset.copy),
  'clear':      b => { b.closest('.fld').querySelector('input').value = ''; },
};
document.addEventListener('click', e => {
  const b = e.target.closest('[data-action]');
  if (b && actions[b.dataset.action]) actions[b.dataset.action](b);
});
