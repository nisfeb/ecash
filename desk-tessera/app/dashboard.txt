const API = "/apps/tessera/api";
async function req(path, opts) {
  opts = opts || {};
  const headers = opts.body ? {'Content-Type': 'application/json'} : {};
  const r = await fetch(API + path, {credentials: 'same-origin', ...opts, headers});
  let data;
  try { data = await r.json(); } catch { data = null; }
  if (!r.ok) throw new Error((data && data.detail) || ('HTTP ' + r.status));
  return data;
}
const post = (path, body) => req(path, {method: 'POST', body: JSON.stringify(body)});
const el = id => document.getElementById(id);
const esc = s => String(s == null ? '' : s).replace(/&/g, '&amp;').replace(/</g, '&lt;')
  .replace(/>/g, '&gt;').replace(/'/g, '&#39;').replace(/"/g, '&quot;');
function toast(msg) { const t = el('toast'); t.textContent = msg; t.style.opacity = '1'; setTimeout(() => { t.style.opacity = '0'; }, 2600); }
function status(ok) { const s = el('sta'); s.textContent = ok ? 'Online' : 'Error'; s.style.color = ok ? '#5eead4' : '#fca5a5'; }
const box = (v, l) => '<div class="box"><div class="num">' + esc(v) + '</div><div class="lbl">' + esc(l) + '</div></div>';
const run = (p, then) => p.then(then).catch(e => toast(e.message));
const list = s => s.split(/[\s,]+/).map(x => x.trim()).filter(Boolean);
const ranks = {galaxy: 'galaxies', star: 'stars', planet: 'planets', moon: 'moons', comet: 'comets'};
const dur = s => s == null ? '' : s % 86400 === 0 ? (s / 86400) + 'd' : s % 3600 === 0 ? (s / 3600) + 'h' : s % 60 === 0 ? (s / 60) + 'm' : s + 's';
// "30m", "1d", "90" (seconds) -> seconds; blank -> null
function secs(v, what) {
  v = v.trim();
  if (v === '') return null;
  const m = /^(\d+)\s*([smhd]?)$/.exec(v);
  if (!m || +m[1] === 0) throw new Error(what + ': a length like 90, 30m, 12h or 1d');
  return +m[1] * {'': 1, s: 1, m: 60, h: 3600, d: 86400}[m[2]];
}
function whole(v, what) {
  v = v.trim();
  if (v === '') return null;
  if (!/^\d+$/.test(v)) throw new Error(what + ': a whole number');
  return +v;
}
function when(v) {
  if (v === '') return null;
  const n = Math.floor(new Date(v).getTime() / 1000);
  if (!(n > 0)) throw new Error('Expires: a date');
  return n;
}
function localInput(unix) {
  const d = new Date(unix * 1000);
  d.setMinutes(d.getMinutes() - d.getTimezoneOffset());
  return d.toISOString().slice(0, 16);
}

  document.querySelectorAll('#tabs .tab').forEach(x => x.classList.remove('on'));
  document.querySelectorAll('.sec').forEach(s => s.classList.remove('on'));
  b.classList.add('on');
  el('sec-' + b.dataset.tab).classList.add('on');
  ({ov: loadOverview, sv: loadServices, wl: loadWallet})[b.dataset.tab]();
}));

async function loadOverview() {
  try {
    const [o, s] = await Promise.all([req('/overview'), req('/services')]);
    const svcs = s.services || [];
    el('ov-stats').innerHTML =
      box(svcs.filter(x => x.active).length + ' / ' + svcs.length, 'Services active') +
      box(svcs.reduce((a, x) => a + x.issued, 0), 'Tokens issued') +
      box(svcs.reduce((a, x) => a + x.redeemed, 0), 'Tokens used') +
      box(o.spent, 'Spent, kept') +
      box(o.quota_counts, 'Quota counts') +
      box(o.held, 'Tokens held') +
      box(o.pending, 'Requests out');
    status(true);
  } catch (e) { status(false); toast(e.message); }
}

function who(s) {
  const g = [];
  if (s.open) g.push('anyone');
  if (s.keys.length) g.push(s.keys.length + ' key' + (s.keys.length === 1 ? '' : 's'));
  if (s.ships.length) g.push(s.ships.length + ' ship' + (s.ships.length === 1 ? '' : 's'));
  if (s.rank) g.push(ranks[s.rank] + ' and up');
  return g.length ? g.join(', ') : 'this ship only';
}

function card(s) {

  const expired = s.expires != null && s.expires < Date.now() / 1000;
  const f = (label, cls, value, hint, extra) => '<div class="fld"><label>' + label + '</label><input class="' + cls + '" value="' + esc(value) + '"' + (extra || '') + '>' + (hint ? '<div class="hint">' + hint + '</div>' : '') + '</div>';
  return '<div class="svc" data-name="' + n + '">' +
    '<div style="display:flex;justify-content:space-between;gap:8px;flex-wrap:wrap"><div>' +
      '<span class="nm">' + n + '</span>' +
      '<span class="badge ' + (s.active && !expired ? 'b-on">active' : 'b-off">' + (expired ? 'expired' : 'inactive')) + '</span>' +
      '<span class="badge b-gate">' + esc(who(s)) + '</span>' +
      '<div>' + esc(s.title) + ' <span class="muted">' + esc(s.description) + '</span></div></div>' +
      '<div style="display:flex;gap:6px;flex-wrap:wrap">' +
        (s.active ? '<button class="rbtn" data-action="deact">Deactivate</button>'
                  : '<button class="rbtn" data-action="act">Activate</button><button class="rbtn" data-action="del">Delete</button>') +
        '<button class="rbtn" data-action="edit">Edit</button><button class="rbtn" data-action="batch">Batch</button>' +
      '</div></div>' +
    '<div class="meta">' +
      '<span>issued <b>' + s.issued + (s.max_issuance != null ? ' / ' + s.max_issuance : '') + '</b></span>' +
      '<span>used <b>' + s.redeemed + '</b></span>' +
      '<span>mode <b>' + esc(s.mode) + '</b></span>' +
      (s.quota ? '<span>quota <b>' + s.quota.n + ' per ' + dur(s.quota.per) + '</b></span>' : '') +
      (s.window ? '<span>tokens last <b>to the end of each ' + dur(s.window) + '</b></span>' : '') +
      (s.expires != null ? '<span>' + (expired ? 'expired ' : 'expires ') + new Date(s.expires * 1000).toLocaleString() + '</span>' : '') +
      '<span class="mono muted">/tessera/' + n + '</span>' +
    '</div>' +
    '<div class="edit" hidden>' +
      '<div class="frow">' + f('Title', 'e-title', s.title) + f('Description', 'e-desc', s.description) + '</div>' +
      '<div class="frow">' +
        '<div class="fld"><label>Anyone may get tokens</label><select class="e-open"><option value="0"' + (s.open ? '' : ' selected') + '>no</option><option value="1"' + (s.open ? ' selected' : '') + '>yes</option></select></div>' +
        '<div class="fld"><label>Or ships of rank</label><select class="e-rank">' +
          ['', 'galaxy', 'star', 'planet', 'moon', 'comet'].map(r => '<option value="' + r + '"' + ((s.rank || '') === r ? ' selected' : '') + '>' + (r ? ranks[r] + ' and up' : 'none') + '</option>').join('') + '</select></div>' +
        '<div class="fld"><label>Mode</label><select class="e-mode"><option value="burn"' + (s.mode === 'burn' ? ' selected' : '') + '>burn: a token is used once</option><option value="check"' + (s.mode === 'check' ? ' selected' : '') + '>check: a token can be shown again</option></select></div>' +
      '</div>' +
      '<div class="frow">' + f('Ships', 'e-ships', s.ships.join(' '), 'over Ames, space or comma separated') +
        f('Client keys (Clear-auth)', 'e-keys', s.keys.join(' '), 'for web clients') + '</div>' +
      '<div class="frow">' + f('Verifier ships', 'e-vers', s.verifiers.join(' '), 'may check or burn tokens they are shown') +
        f('Verifier keys', 'e-vkeys', s.verifier_keys.join(' '), 'for resource servers on /redeem and /check') +
        f('Agents', 'e-agents', s.agents.join(' '), 'on this ship, that a presented token may be delivered to') + '</div>' +
      '<div class="frow">' +
        f('Quota: tokens', 'e-qn', s.quota ? s.quota.n : '', 'per identity; blank for none') +

        f('Max issued', 'e-max', s.max_issuance == null ? '' : s.max_issuance, 'blank: no cap') +

async function loadServices() {
  try {
    const svcs = ((await req('/services')).services || []).sort((a, b) => a.name.localeCompare(b.name));
    el('svc-list').innerHTML = svcs.length ? svcs.map(card).join('') : '<div class="empty">No services yet.</div>';
    status(true);
  } catch (e) { status(false); el('svc-list').innerHTML = '<div class="empty">' + esc(e.message) + '</div>'; }
}

function save(c) {
  const q = cls => c.querySelector('.' + cls).value;
  try {
    const qn = whole(q('e-qn'), 'Quota'), per = secs(q('e-qper'), 'Quota per');
    if ((qn == null) !== (per == null)) throw new Error('Quota: both a number and a length, or neither');
    const body = {
      name: c.dataset.name, title: q('e-title').trim(), description: q('e-desc'),
      open: q('e-open') === '1', rank: q('e-rank') || null, mode: q('e-mode'),
      ships: list(q('e-ships')), keys: list(q('e-keys')),
      verifiers: list(q('e-vers')), verifier_keys: list(q('e-vkeys')), agents: list(q('e-agents')),
      quota: qn == null ? null : {n: qn, per}, window: secs(q('e-win'), 'Window'),
      max_issuance: whole(q('e-max'), 'Max issued'), expires: when(q('e-exp')),
    };
    run(post('/services/update', body), () => { toast('Saved'); loadServices(); });
  } catch (e) { toast(e.message); }
}

function batch(c) {
  const n = prompt('How many tokens to make (1-100)? They count toward the cap.', '10');
  if (!n) return;
  run(post('/services/batch', {name: c.dataset.name, n: +n}), d => {
    const out = c.querySelector('.out');
    out.hidden = false;
    out.textContent = d.tokens.join('\n');
    loadServices();
  });
}

async function loadWallet() {
  try {
    const d = await req('/wallet');
    const w = d.wallet || [], o = d.offers || [];
    el('wl-list').innerHTML = (w.length
      ? '<div class="tbl-wrap"><table class="tbl"><thead><tr><th>Issuer</th><th>Service</th><th>Tokens</th></tr></thead><tbody>' +
        w.map(x => '<tr><td class="mono">' + esc(x.issuer) + '</td><td class="accent">' + esc(x.service) + '</td><td>' + x.count + '</td></tr>').join('') +
        '</tbody></table></div>'
      : '<div class="empty">No tokens held.</div>') +
      (o.length
      ? '<div class="ch" style="border:none;padding-left:0">Offered to this ship</div><div class="tbl-wrap"><table class="tbl"><thead><tr><th>From</th><th>Issuer</th><th>Service</th><th>Tokens</th><th></th></tr></thead><tbody>' +
        o.map(x => '<tr data-from="' + esc(x.from) + '" data-issuer="' + esc(x.issuer) + '" data-service="' + esc(x.service) + '">' +
          '<td class="mono">' + esc(x.from) + '</td><td class="mono">' + esc(x.issuer) + '</td><td class="accent">' + esc(x.service) + '</td><td>' + x.count + '</td>' +
          '<td><button class="rbtn" data-action="w-accept">Accept</button> <button class="rbtn" data-action="w-decline">Decline</button></td></tr>').join('') +
        '</tbody></table></div>'
      : '');
    status(true);
  } catch (e) { status(false); el('wl-list').innerHTML = '<div class="empty">' + esc(e.message) + '</div>'; }
}

function wallet(act) {
  const body = {issuer: el('w-issuer').value.trim(), service: el('w-svc').value.trim()};
  try {
    if (act === 'get' || act === 'give') body.n = +el('w-n').value;
    if (act === 'give') body.to = el('w-to').value.trim();
    if (act === 'use' && el('w-agent').value.trim()) {
      body.agent = el('w-agent').value.trim();
      const d = el('w-data').value.trim();
      if (d) body.data = JSON.parse(d);
    }
  } catch (e) { toast('Data: ' + e.message); return; }
  const out = el('w-out');
  out.hidden = false;
  out.textContent = act + '...';
  post('/wallet/' + act, body)
    .then(d => { out.textContent = JSON.stringify(d); loadWallet(); })
    .catch(e => { out.textContent = e.message; });
}

const act = {
  'refresh-ov': loadOverview, 'refresh-sv': loadServices, 'refresh-wl': loadWallet,
  'prune': () => run(post('/prune', {}), () => { toast('Pruned'); loadOverview(); }),
  'create': () => {
    const name = el('new-name').value.trim(), title = el('new-title').value.trim();
    if (!/^[a-z0-9_-]{1,64}$/.test(name)) { toast('Name: 1-64 of a-z, 0-9, _ and -'); return; }
    if (!title) { toast('A title, please'); return; }
    run(post('/services/create', {name, title}), () => { el('new-name').value = ''; el('new-title').value = ''; loadServices(); });
  },
  'edit': c => { const e = c.querySelector('.edit'); e.hidden = !e.hidden; },
  'save': save,
  'batch': batch,
  'act': c => run(post('/services/activate', {name: c.dataset.name}), loadServices),
  'deact': c => confirm('Deactivate ' + c.dataset.name + '? Its tokens stop working until it is back.') &&
    run(post('/services/deactivate', {name: c.dataset.name}), loadServices),
  'del': c => confirm('Delete ' + c.dataset.name + '? Only a service that never issued can go.') &&
    run(post('/services/delete', {name: c.dataset.name}), loadServices),
  'w-get': () => wallet('get'), 'w-use': () => wallet('use'), 'w-take': () => wallet('take'), 'w-give': () => wallet('give'),
  'w-accept': b => offer('accept', b), 'w-decline': b => offer('decline', b),
};
// accept an offer (a refresh at its issuer) or decline it
function offer(act, b) {
  const r = b.closest('tr').dataset;
  run(post('/wallet/' + act, {from: r.from, issuer: r.issuer, service: r.service}), d => { toast((act === 'accept' ? 'Accepted ' : 'Declined ') + d.done); loadWallet(); });
}
