const ADMIN = "/apps/ecash/admin/api";
const PAGE = 50;  // quotes rendered per page
let allQuotes = [], currentFilter = 'all', qPage = 0;
let activeKeysetId = '';  // from /overview; the keyset the mint signs with
let curKs = null;         // keyset shown in the detail view
let selfWas = false;      // self_method_enabled as last read from / saved to the server

// ── Helpers ──────────────────────────────────────────────────
// Read the body ONCE (a second read throws "body stream already read"). Errors are
// JSON {detail} normally, but an Eyre/proxy error page may not be JSON at all.
async function adminApi(path, opts) {
  const r = await fetch(ADMIN + path, opts);
  const text = await r.text();
  let body = null;
  try { body = JSON.parse(text); } catch {}
  if (!r.ok) throw new Error(body && body.detail != null ? String(body.detail) : "HTTP " + r.status);
  if (!body || typeof body !== 'object') throw new Error("HTTP " + r.status + ": unexpected non-JSON response");
  return body;
}
const post = (p, b) => adminApi(p, {
  method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify(b || {})
});

const esc = (s) => String(s == null ? '' : s)
  .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
  .replace(/'/g, '&#39;').replace(/"/g, '&quot;');

const errBox = (e) => `<div class="res err" style="display:block">${esc(e.message)}</div>`;

function box(val, label, cls) {
  const isLong = String(val).length > 16;
  return `<div class="box"><div class="num ${isLong ? 'sm' : ''} ${cls || ''}">${esc(val)}</div><div class="lbl">${esc(label)}</div></div>`;
}

function badge(text, type) {
  return `<span class="badge b-${type}">${esc(text)}</span>`;
}

function fmtDate(unix) {
  if (!unix) return '?';
  return new Date(unix * 1000).toLocaleString();
}

function truncId(id, n) {
  if (!id) return '?';
  id = String(id);
  n = n || 16;
  return id.length > n ? id.slice(0, n) + '...' : id;
}

// A keyboard-reachable "click to copy" value. n = truncate length (0 = full).
const copyBtn = (v, n, cls) =>
  `<button class="lnk mono ${cls || ''}" data-act="copyText" data-arg="${esc(v)}" title="Copy ${esc(v)}">${esc(n ? truncId(v, n) : v)}</button>`;

function copyText(t) {
  if (navigator.clipboard) navigator.clipboard.writeText(t).then(() => toast('Copied!'), () => copyFallback(t));
  else copyFallback(t);
}
// navigator.clipboard only exists in secure contexts (not plain HTTP on a LAN IP).
function copyFallback(t) {
  const ta = document.createElement('textarea');
  ta.value = t;
  ta.style.cssText = 'position:fixed;top:0;left:0;opacity:0';
  document.body.appendChild(ta);
  ta.select();
  let ok = false;
  try { ok = document.execCommand('copy'); } catch {}
  ta.remove();
  if (ok) toast('Copied!'); else prompt('Copy this value:', t);
}

let toastTimer;
function toast(msg) {
  const el = document.getElementById("toast");
  el.textContent = msg;
  el.style.opacity = "1";
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => { el.style.opacity = "0"; }, 1500);
}

function showResult(elId, text, type) {
  const el = document.getElementById(elId);
  el.style.display = 'block';
  el.className = 'res' + (type ? ' ' + type : '');
  el.textContent = text;
}

// Strong confirmation for destructive money actions: the operator must type `word`.
function typedConfirm(msg, word) {
  const v = prompt(msg + '\n\nType ' + word + ' to confirm:');
  if (v === null) return false;
  if (v.trim() !== word) { alert('Confirmation did not match. Nothing was done.'); return false; }
  return true;
}

}
function stopAuto() { if (_ar) { clearInterval(_ar); _ar = null; } }

// ── Overview + Stats Bar (one /overview fetch renders both) ──
async function loadOverview(silent) {
  const sta = document.getElementById("sta");
  try {
    const d = await adminApi("/overview");
    activeKeysetId = d.active_keyset || '';

    // Outstanding liability (issued − redeemed). Can go negative after a revoke.
    const issued = Number(d.total_issued_sats) || 0;
    const redeemed = Number(d.total_redeemed_sats) || 0;
    const outstanding = issued - redeemed;

    document.getElementById("stats").innerHTML =
      box(d.tokens_issued || 0, "Tokens Issued") +
      box(d.spent_count || 0, "Tokens Spent") +
      box(issued.toLocaleString(), "Issued (sats)") +
      box(redeemed.toLocaleString(), "Redeemed (sats)") +
      box(outstanding.toLocaleString(), "Outstanding (sats)", outstanding < 0 ? 'neg' : '') +
      box(d.ln_backend?.configured ? d.ln_backend.type : "none", "LN Backend") +
      box(d.pending_requests || 0, "Pending Requests");

    document.getElementById("ov-liability").innerHTML = `
      <div style="display:flex;gap:32px;flex-wrap:wrap;align-items:flex-end">
        <div>
          <div class="num ${outstanding < 0 ? 'neg' : ''}">${esc(outstanding.toLocaleString())}</div>
          <div class="lbl">Outstanding (sats)</div>
        </div>
        <div>
          <div class="num sm" style="font-size:1.1rem">${esc(issued.toLocaleString())}</div>
          <div class="lbl">Issued (sats)</div>
        </div>
        <div>
          <div class="num sm" style="font-size:1.1rem">${esc(redeemed.toLocaleString())}</div>
          <div class="lbl">Redeemed (sats)</div>
        </div>
      </div>
      <div class="hint" style="margin-top:8px">Outstanding = issued − redeemed. This is the mint's live token liability.</div>
      ${outstanding < 0 ? '<div class="hint" style="color:#fca5a5">Negative: more was redeemed than is recorded as issued (e.g. tokens from a revoked ISSUED quote were redeemed after all).</div>' : ''}
    `;

async function updateSettings() {
  const pct = parseInt(document.getElementById("st-fee-pct").value);
  const min = parseInt(document.getElementById("st-fee-min").value);

async function setKeysetFee() {
  const ks = curKs;
  if (!ks) return;
  const currentFee = ks.input_fee_ppk || 0;
  const input = prompt('Set input fee (parts per thousand) for keyset:\n' + ks.id + '\n\nCurrent fee: ' + currentFee + ' ppk', currentFee);
  if (input === null) return;
  const fee = parseInt(input, 10);
  if (isNaN(fee) || fee < 0) { alert('Invalid fee value'); return; }
  if (fee !== currentFee && isCurrentKs(ks) && !confirm(
    'This is the ACTIVE keyset. Changing its fee ROTATES the mint to fresh keys under a NEW keyset id and deactivates this one.\n\n' +
    'Tokens already issued under the old id still redeem, but wallets must refetch the mint\'s keys before they can mint or swap against the new keyset.\n\nContinue?')) return;
  try {
    const d = await post("/keysets/set-fee", {id: ks.id, input_fee_ppk: fee});
    if (d.old_id === d.new_id) {
      showResult("ks-action-res", "Fee unchanged (" + d.input_fee_ppk + " ppk): nothing to do, keyset id kept.", "ok");
    } else {
      showResult("ks-action-res", "Fee set to " + d.input_fee_ppk + " ppk under fresh keys.\nNew keyset id: " + d.new_id +
        "\nOld id " + d.old_id + " is now inactive; tokens issued under it still redeem.", "ok");
    }
    await refreshKeysets();
    viewKeyset(d.new_id);
  } catch(e) {
    showResult("ks-action-res", "Error setting fee: " + e.message, "err");
  }
}

// ── Quotes Tab ──────────────────────────────────────────────
const STATE_CLS = {PAID: 'paid', ISSUED: 'issued', PENDING: 'pending'};

async function loadQuotes() {
  try {
    const d = await adminApi("/quotes");
    allQuotes = [];
    (d.mint_quotes || []).forEach(q => allQuotes.push({...q, _type: 'mint'}));
    (d.melt_quotes || []).forEach(q => allQuotes.push({...q, _type: 'melt'}));
    allQuotes.sort((a,b) => (b.created||0) - (a.created||0));
    renderQuotes();
  } catch(e) {
    document.getElementById("qt-body").innerHTML = errBox(e);
  }
}

// Offer only what the backend accepts: delete refuses PAID/ISSUED mint quotes and
// PAID/PENDING melts; revoke is the mint-quote override; abort is for PENDING bolt11 melts.
function quoteActions(q) {
  const id = esc(q.quote_id), out = [];
  const mint = q._type === 'mint';
  if (!(mint ? ['PAID', 'ISSUED'] : ['PAID', 'PENDING']).includes(q.state))
    out.push(`<button class="rbtn rbtn-bad" data-act="deleteQuote" data-arg="${id}" data-arg2="${q._type}">Delete</button>`);
  if (mint && (q.state === 'PAID' || q.state === 'ISSUED'))
    out.push(`<button class="rbtn rbtn-bad" data-act="revokeQuote" data-arg="${id}">Revoke</button>`);
  if (!mint && q.state === 'PENDING' && q.method !== 'self')
    out.push(`<button class="rbtn" data-act="abortMelt" data-arg="${id}">Abort</button>`,
             `<button class="rbtn rbtn-bad" data-act="forceAbortMelt" data-arg="${id}">Force abort</button>`);
  return out.join(' ');
}

function renderQuotes() {
  const filtered = currentFilter === 'all' ? allQuotes
    : (currentFilter === 'mint' || currentFilter === 'melt') ? allQuotes.filter(q => q._type === currentFilter)
    : allQuotes.filter(q => q.state === currentFilter);

  if (!filtered.length) {
    document.getElementById("qt-body").innerHTML = '<div class="empty">No quotes match filter</div>';
    return;
  }

  const pages = Math.ceil(filtered.length / PAGE);
  qPage = Math.max(0, Math.min(qPage, pages - 1));
  const rows = filtered.slice(qPage * PAGE, qPage * PAGE + PAGE);

  document.getElementById("qt-body").innerHTML = `
    <div class="tw"><table class="tbl">
      <thead><tr>
        <th>Type</th><th>Quote ID</th><th>Amount</th><th>State</th><th>Method</th><th>Payment hash</th><th>Created</th><th>Expired</th><th></th>
      </tr></thead>
      <tbody>
        ${rows.map(q => `<tr>
            <td>${badge(q._type, q._type === 'mint' ? 'active' : 'pending')}</td>
            <td>${copyBtn(q.quote_id, 16)}</td>
            <td><strong>${esc(q.amount ?? '?')}</strong> sats</td>
            <td>${badge(q.state, STATE_CLS[q.state] || 'unpaid')}</td>
            <td>${esc(q.method || '?')}</td>
            <td>${q.payment_hash ? copyBtn(q.payment_hash, 12) : '<span class="muted">-</span>'}</td>
            <td class="muted">${esc(fmtDate(q.created))}</td>
            <td>${q.expired ? badge('Yes', 'expired') : '<span class="muted">No</span>'}</td>
            <td style="white-space:nowrap">${quoteActions(q)}</td>
          </tr>`).join("")}
      </tbody>
    </table></div>
    <div class="hint" style="margin-top:8px;display:flex;gap:8px;align-items:center;flex-wrap:wrap">
      ${pages > 1 ? `<button class="rbtn" data-act="quotePage" data-arg="-1"${qPage ? '' : ' disabled'}>&larr; Prev</button>
      <span>Page ${qPage + 1} of ${pages}</span>
      <button class="rbtn" data-act="quotePage" data-arg="1"${qPage < pages - 1 ? '' : ' disabled'}>Next &rarr;</button>` : ''}
      <span>Showing ${rows.length} of ${filtered.length} (${allQuotes.length} total). Click an ID or hash to copy.</span>
    </div>
  `;
}

function quotePage(delta) { qPage += Number(delta); renderQuotes(); }

const findQuote = (id, type) => allQuotes.find(q => q.quote_id === id && q._type === type);

// Reload after every quote action, success or not: a refusal usually means the state moved.
async function afterQuoteAction() {
  await loadQuotes();
  loadOverview(true);
}

async function deleteQuote(quoteId, type) {
  if (!confirm('Delete this ' + type + ' quote?\n\nID: ' + quoteId)) return;
  try {
    await post("/quotes/delete", {quote_id: quoteId, type});
    showResult("qt-res", "Deleted " + type + " quote " + quoteId, "ok");
  } catch(e) {
    showResult("qt-res", "Delete failed: " + e.message, "err");
  }
  await afterQuoteAction();
}

async function revokeQuote(quoteId) {
  const q = findQuote(quoteId, 'mint') || {};
  const amt = (q.amount ?? '?') + ' sats';
  const ok = q.state === 'PAID'
    ? typedConfirm('REVOKE a PAID mint quote?\n\n' +
        'The customer has PAID ' + amt + ' for this quote but has NOT minted their tokens yet. ' +
        'Revoking deletes the quote: they can never mint, and their deposit is destroyed as far as the mint is concerned.\n\n' +
        'Only do this if you have refunded them some other way or know the payment is bogus.\n\nID: ' + quoteId, 'REVOKE')
    : confirm('Revoke this ' + (q.state || '') + ' mint quote?\n\n' +
        'The quote is deleted' + (q.state === 'ISSUED' ? ' and ' + amt + ' is subtracted from total issued: the tokens minted from it are declared never-to-be-redeemed. ' +
        'If they ARE redeemed later, outstanding liability goes negative.' : '.') + '\n\nID: ' + quoteId);
  if (!ok) return;
  try {
    const d = await post("/quotes/revoke", {quote_id: quoteId});
    showResult("qt-res", "Revoked mint quote " + d.quote_id + " (was " + d.was_state + "); total issued reduced by " +
      d.issued_decremented + " sats.", "ok");
  } catch(e) {
    showResult("qt-res", "Revoke failed: " + e.message, "err");
  }
  await afterQuoteAction();
}

const ABORT_RESULT = {
  'settled-not-aborted': 'Lightning shows the payment SETTLED: the quote was marked PAID instead of aborted.',
  'in-flight-or-unconfirmed': 'Lightning did not confirm a failure (in flight or unknown). Nothing changed; the quote stays PENDING. Force abort only if you KNOW the payment failed.',
  'aborted-confirmed-failed': "Lightning confirmed the payment FAILED: the customer's inputs were un-spent and the quote reset.",
  'aborted-forced': "Force-aborted: the customer's inputs were un-spent on your assertion that the payment failed.",
  'no-op-not-pending': 'The quote is no longer PENDING. Nothing changed.',
  'no-inflight-record': 'No in-flight record to reverse. Nothing changed.',
};

async function abortMelt(quoteId, force) {
  const q = findQuote(quoteId, 'melt') || {};
  const hash = q.payment_hash || '(none recorded)';
  const ok = force
    ? typedConfirm('FORCE-abort this PENDING melt?\n\n' +
        "DANGER: you are asserting the Lightning payment FAILED and authorizing the mint to un-spend the customer's inputs. " +
        'If the payment actually went through (or still goes through), the mint pays TWICE.\n\n' +
        'The mint still re-checks Lightning and settles instead if it sees the payment succeeded, but an in-flight payment ' +
        'looks the same as a failed one. Only force after verifying on your node that this payment FAILED.\n\n' +
        'Payment hash: ' + hash + '\nQuote: ' + quoteId, 'FORCE')
    : confirm('Abort this PENDING melt?\n\nThe mint re-checks the payment on Lightning first:\n' +
        '- settled: the quote is marked PAID (not aborted)\n' +
        "- confirmed failed: the customer's inputs are un-spent and the quote reset\n" +
        '- anything ambiguous: nothing changes\n\nPayment hash: ' + hash + '\nQuote: ' + quoteId);
  if (!ok) return;
  try {
    const d = await post("/melt/abort", force ? {quote_id: quoteId, force: true} : {quote_id: quoteId});
    const msg = ABORT_RESULT[d.result] || (d.aborted
      ? "Aborted without a Lightning re-check: the customer's inputs were un-spent."
      : 'Not aborted' + (d.result ? ': ' + d.result : '.'));
    showResult("qt-res", msg + "\n\nquote " + (d.quote_id || quoteId) + " | aborted: " + !!d.aborted +
      " | Lightning checked: " + (d.ln_checked ?? 'n/a'),
      d.aborted || d.result === 'settled-not-aborted' ? "ok" : "");
  } catch(e) {
    showResult("qt-res", "Abort failed: " + e.message, "err");
  }
  await afterQuoteAction();
}

async function checkSpent() {
  const input = document.getElementById("tk-check-input").value.trim();
  if (!input) return;
  // A compressed point is looked up as Y; anything else is a secret.
  const isY = /^0[23][0-9a-f]{64}$/i.test(input);
  try {
    const d = await post("/spent/check", isY ? {Y: input.toLowerCase()} : {secret: input});
    const what = isY ? "Y-point" : "Secret";
    if (d.spent) {
      showResult("tk-check-res", what + ": SPENT - This token has been redeemed and cannot be used again.", "err");
    } else {
      showResult("tk-check-res", what + ": UNSPENT - This token has not been seen by the mint (or is not tracked).", "ok");
    }
  } catch(e) {
    showResult("tk-check-res", "Error: " + e.message, "err");
  }
}

    } else {

function toggleLnFields() {
  const type = document.getElementById("ln-type").value;
  const fields = document.getElementById("ln-fields");
  const apikeyField = document.getElementById("ln-apikey-field");
  const macaroonField = document.getElementById("ln-macaroon-field");
  if (type === 'none') {
    fields.style.display = 'none';
  } else {
    fields.style.display = 'block';
    apikeyField.style.display = type === 'lnbits' ? 'flex' : 'none';
    macaroonField.style.display = type === 'lnd' ? 'flex' : 'none';
  }
}

async function configureLn() {
  const type = document.getElementById("ln-type").value;
  let body;
  if (type === 'none') {
    if (!confirm('Remove Lightning backend configuration?')) return;
    body = {type: 'none'};
  } else if (type === 'lnbits') {
    const url = document.getElementById("ln-url").value.trim();
    const apiKey = document.getElementById("ln-apikey").value.trim();
    if (!url || !apiKey) { alert('URL and API Key are required'); return; }
    body = {type: 'lnbits', url, api_key: apiKey};
  } else if (type === 'lnd') {
    const url = document.getElementById("ln-url").value.trim();
    const macaroon = document.getElementById("ln-macaroon").value.trim();
    if (!url || !macaroon) { alert('URL and Macaroon are required'); return; }
    body = {type: 'lnd', url, macaroon};
  }
  try {
    const d = await post("/lightning/configure", body);
    showResult("ln-config-res", "Configured: " + d.type + (d.url ? " at " + d.url : ""), "ok");
    await loadLightning();
    loadOverview(true);
  } catch(e) {
    showResult("ln-config-res", "Error: " + e.message, "err");
  }
}

async function clearLn() {
  if (!confirm('Remove Lightning backend? bolt11 minting/melting will be disabled.')) return;
  try {
    await post("/lightning/configure", {type: 'none'});
    showResult("ln-config-res", "Lightning backend removed", "ok");
    await loadLightning();
    loadOverview(true);
  } catch(e) {
    showResult("ln-config-res", "Error: " + e.message, "err");
  }
}

// A real round-trip to the backend: report ok/error exactly as the mint saw it.
async function testLn() {
  showResult("ln-config-res", "Testing...", "");
  try {
    const d = await post("/lightning/test");
    const ok = d.status === 'ok';
    const lines = [
      String(d.status || '?').toUpperCase() + ": " + (d.type || '?') + (d.url ? " at " + d.url : ""),
      d.http_status != null ? "HTTP status: " + d.http_status : "",
      d.balance_msat != null ? "Balance: " + Math.floor(Number(d.balance_msat) / 1000).toLocaleString() + " sats" : "",
      d.detail != null ? "Detail: " + (typeof d.detail === 'string' ? d.detail : JSON.stringify(d.detail)) : "",
    ];
    showResult("ln-config-res", lines.filter(Boolean).join("\n"), ok ? "ok" : "err");
  } catch(e) {
    showResult("ln-config-res", "Test failed: " + e.message, "err");
  }
}

async function updateInfo() {
  const name = document.getElementById("nf-name").value.trim();
  const description = document.getElementById("nf-desc").value.trim();
  if (!name && !description) { alert('Enter a name or description'); return; }
  try {
    const body = {};
    if (name) body.name = name;
    if (description) body.description = description;
    const d = await post("/info/update", body);
    showResult("nf-res", "Updated: " + d.name + " - " + d.description, "ok");
    await loadInfo();
  } catch(e) {
    showResult("nf-res", "Error: " + e.message, "err");
  }
}

// Lazy-load tabs on first visit
const loaded = {ov: true, ks: false, qt: false, tk: false, ln: false, nf: false};
document.querySelectorAll("#main-tabs .tab").forEach(btn => {
  btn.addEventListener("click", async () => {
    const t = btn.dataset.tab;
    if (loaded[t]) return;
    loaded[t] = true;
    if (t === 'ks') await loadKeysets();
    if (t === 'qt') await loadQuotes();
    if (t === 'tk') await loadSpent();
    if (t === 'ln') await loadLightning();
    if (t === 'nf') await loadInfo();
  });
});

// Event delegation: CSP blocks inline on*= handlers, so elements carry data-act
// (+ optional data-arg/data-arg2) and we resolve them to functions here. Args are
// data, never code — strictly safer than the old inline onclick handlers.
const ACTIONS = {
  loadOverview, updateSettings, generateKeyset, loadKeysets, showKeysetList,
  loadQuotes, loadSpent, checkSpent, loadLightning, toggleLnFields, configureLn,
  testLn, loadInfo, updateInfo, clearLn, copyText, viewKeyset, activateKeyset,
  deactivateKeyset, setKeysetFee, deleteQuote, revokeQuote, abortMelt, quotePage,
  forceAbortMelt: (id) => abortMelt(id, true),
};
document.addEventListener('click', (e) => {
  const el = e.target.closest('[data-act]');
  if (!el) return;
  const fn = ACTIONS[el.dataset.act];
  if (!fn) return;
  if ('arg2' in el.dataset) fn(el.dataset.arg, el.dataset.arg2);
  else if ('arg' in el.dataset) fn(el.dataset.arg);
  else fn();
});
document.addEventListener('change', (e) => {
  const el = e.target.closest('[data-change]');
  if (!el) return;
  const fn = ACTIONS[el.dataset.change];
  if (fn) fn();
});

init();
